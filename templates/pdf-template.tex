% Colors
\definecolor{fivb-blue}{RGB}{0,51,153}

% Fonts
% Use Inter everywhere (bundled into docs/assets so GitHub Pages serves it too).

  Path=docs/assets/fonts/Inter/,

  colorlinks=true,

  pdfborder={0 0 0}

% Custom commands for rules
% NOTE: Don't name this \rule because LaTeX already defines \rule.
\newcommand{\RuleBlock}[2]{%
  \phantomsection\label{rule-#1}%
  \noindent\rule{\textwidth}{0.5pt}\\[0.5em]
  {\color{fivb-blue}\textbf{\large #1 #2}}\\[0.5em]
}

\newcommand{\SubruleBlock}[2]{%
  \phantomsection\label{rule-#1}%
  \vspace{1.2em}
  \noindent\textbf{#1 #2}\\[0.3em]
}

\newcommand{\ruleitem}[2]{%
  \phantomsection\label{rule-#1}%
  \hangindent=1em\hangafter=1%
  \textbf{#1} \hspace{0.5em} #2
}

\newcommand{\DiagramBlock}[2]{%
  \phantomsection\label{rule-D#1}%
  \noindent\rule{\textwidth}{0.5pt}\\[0.5em]
  {\color{fivb-blue}\textbf{\large KAAVIO #1: #2}}\\[0.5em]
}

% Margin note styling
\renewcommand{\marginnote}[1]{%
  \marginpar{%
    \raggedright%
    \footnotesize%
    \color{fivb-blue}%
    \textbf{#1}%
  }%
}

  {0em}
